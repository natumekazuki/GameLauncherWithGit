private static bool IsMissingGitConfig(GitCommandResult result)
{
	return result.ExitCode == 1
		&& string.IsNullOrWhiteSpace(result.StandardError)
		&& string.IsNullOrWhiteSpace(result.StandardOutput);
}

// 例: remote 取得時
