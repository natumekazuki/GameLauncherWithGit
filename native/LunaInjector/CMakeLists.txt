cmake_minimum_required(VERSION 3.20)
project(GameLauncherLunaInjector LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT DEFINED LUNA_SOURCE_DIR)
  message(FATAL_ERROR "LUNA_SOURCE_DIR is required")
endif()
add_executable(LunaInjector main.cpp)
target_compile_definitions(LunaInjector PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(LunaInjector PRIVATE /utf-8)
target_include_directories(LunaInjector PRIVATE "${LUNA_SOURCE_DIR}/src/NativeImpl/LunaSubprocess")
target_link_libraries(LunaInjector PRIVATE Shlwapi Psapi)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set_target_properties(LunaInjector PROPERTIES OUTPUT_NAME LunaInjector64)
else()
  set_target_properties(LunaInjector PROPERTIES OUTPUT_NAME LunaInjector32)
endif()
